	tasks:
    - apt:
        name: nginx
        state: present

        name: nginx

Install & Configure Tomcat
	
	---
	- name: Install Tomcat
	hosts: app
	become: yes

	tasks:
    - name: Install Java
      apt:
        name: openjdk-11-jdk
        state: present

    - name: Download Tomcat
      get_url:
        url: https:tomcat-9---link address
        dest: /opt/tomcat.tar.gz

    - name: Extract Tomcat
      unarchive:
        src: /opt/tomcat.tar.gz
        dest: /opt/
		
		
		
Using Ansible Roles

	---
	- name: Install nginx
	apt:
	name: nginx
    state: present

	- name: Start nginx
	service:
    name: nginx
    state: started
    enabled: yes
	
	
	---
	- name: Configure servers using roles
	hosts: web
	become: yes

	roles:
    - nginx
	
	
	
	
Create User & Setup SSH Keys
	---
	- name: Create user and setup SSH
	hosts: all
	become: yes

	tasks:
    - name: Create user
      user:
        name: shahbazz
        state: present

    - name: Add SSH key
      authorized_key:
        user: shahbazz
        key: "{{ lookup('file','/.ssh/id_rsa.pub') }}"

Deploy Sample Application on Tomcat
	---
	- name: Deploy WAR file
	hosts: app
	become: yes

	tasks:
    - name: Copy WAR file
      copy:
        src: sample.war
        dest: /opt/tomcat/webapps/

    - name: Restart Tomcat
      shell: /opt/tomcat/bin/shutdown.sh && /opt/tomcat/bin/startup.sh

Playbook with Handlers 

	---
	- name: nginx with handler
	hosts: web
	become: yes

	tasks:
    - name: Copy nginx config
      copy:
        src: nginx.conf
        dest: /etc/nginx/nginx.conf
      notify: Restart nginx

	handlers:
    - name: Restart nginx
      service:
        name: nginx
        state: restarted




Manage Cron Jobs
	
	---
	- name: Setup cron job
	hosts: all
	become: yes

	tasks:
    - name: Add cron job
      cron:
        name: "Backup Job"
        minute: "0"
        hour: "2"
        job: "/usr/bin/backup.sh"
		



Set System Hostname Dynamically
	---
	- name: Set hostname
	hosts: all
	become: yes

	tasks:
    - name: Set hostname
      hostname:
        name: "{{ inventory_hostname }}"
		
		
10 Important Ansible Modules
Module	Description

	apt / yum	Install packages
	service		Manage services
	copy		Copy files 
	file		Manage file 
	user		Manage users
	cron		Manage cron jobs
	get_url		Download files
	unarchive	Extract tar/zip files
	authorized_key	Manage SSH keys
	shell/command	Run commands



Sample deployment.yaml

	apiVersion: apps/v1
	kind: Deployment
	metadata:
	name: nginx-deployment
	spec:
	replicas: 2
	selector:
    matchLabels:
      app: nginx
	template:
    metadata:
      labels:
        app: nginx
    spec:
      containers:
      - name: nginx
        image: nginx:latest
        ports:
        - containerPort: 80
		
		
		
Sample service.yaml
	
	apiVersion: v1
	kind: Service
	metadata:
	name: nginx-service
	spec:
	type: NodePort
	selector:
    app: nginx
	ports:
    - port: 80
      targetPort: 80
      nodePort: 30007
